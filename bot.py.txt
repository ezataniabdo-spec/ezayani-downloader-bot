
import os, re, asyncio, tempfile, logging
from telegram import Update, InlineKeyboardButton, InlineKeyboardMarkup
from telegram.ext import Application, CommandHandler, MessageHandler, CallbackQueryHandler, ContextTypes, filters
import yt_dlp

logging.basicConfig(level=logging.INFO)
BOT_TOKEN = os.getenv("BOT_TOKEN") or os.getenv("TELEGRAM_BOT_TOKEN") or "YOUR_TOKEN_HERE"
CHANNEL_LINK = "https://t.me/ezayani"  # بدلها بقناتك
AD_TEXT = "اشترك فالقناة باش تكمل التحميل"

# --- Download logic with yt-dlp ---
def download_media(url: str, mode="video"):
    ydl_opts_video = {
        "format": "bestvideo+bestaudio/best",
        "outtmpl": "%(id)s.%(ext)s",
        "noplaylist": True,
        "quiet": True,
        "no_warnings": True,
        "merge_output_format": "mp4",
    }
    ydl_opts_audio = {
        "format": "bestaudio/best",
        "outtmpl": "%(id)s.%(ext)s",
        "noplaylist": True,
        "quiet": True,
        "no_warnings": True,
        "postprocessors": [{"key": "FFmpegExtractAudio", "preferredcodec": "mp3", "preferredquality": "192"}],
    }
    opts = ydl_opts_audio if mode=="audio" else ydl_opts_video
    with tempfile.TemporaryDirectory() as tmp:
        opts["outtmpl"] = os.path.join(tmp, "%(id)s.%(ext)s")
        with yt_dlp.YoutubeDL(opts) as ydl:
            info = ydl.extract_info(url, download=True)
            # find file
            for f in os.listdir(tmp):
                path = os.path.join(tmp, f)
                # copy to persistent temp
                dest = os.path.join(tempfile.gettempdir(), f)
                import shutil; shutil.copy(path, dest)
                return dest, info.get("title","video")
    return None, None

# --- Ad Gate ---
user_passed = set()

async def start(update: Update, ctx: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text("🚀 صيفط رابط تيكتوك، انستغرام، يوتيوب...")

async def handle_link(update: Update, ctx: ContextTypes.DEFAULT_TYPE):
    url = update.message.text.strip()
    if not re.search(r"https?://", url):
        return
    ctx.user_data["url"] = url
    # show ad gate - mandatory 10s
    kb = [
        [InlineKeyboardButton("📢 اشترك فالقناة (إجباري)", url=CHANNEL_LINK)],
        [InlineKeyboardButton("✅ تابعت - كمل التحميل (10s)", callback_data="check_ad")]
    ]
    await update.message.reply_text(
        f"{AD_TEXT}\n\n⏳ خاصك تشوف الإعلان 10 ثواني باش يتفتح التحميل\nرابطك: {url[:50]}...",
        reply_markup=InlineKeyboardMarkup(kb)
    )

async def check_ad(update: Update, ctx: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    # enforce 10 sec wait
    url = ctx.user_data.get("url")
    if not url:
        await query.edit_message_text("صيفط الرابط مرة أخرى")
        return
    # check if user already waited? Simple: always wait 10s now
    await query.edit_message_text("⏳ كنساين 10 ثواني... متخرجش")
    await asyncio.sleep(10)
    user_id = query.from_user.id
    user_passed.add(user_id)
    kb = [
        [InlineKeyboardButton("⬇️ تحميل HD", callback_data="dl_hd"),
         InlineKeyboardButton("🎵 تحميل MP3", callback_data="dl_mp3")],
        [InlineKeyboardButton("📤 Share", switch_inline_query=url)]
    ]
    await query.edit_message_text("✅ تم! اختار نوع التحميل:", reply_markup=InlineKeyboardMarkup(kb))

async def do_download(update: Update, ctx: ContextTypes.DEFAULT_TYPE, mode):
    query = update.callback_query
    url = ctx.user_data.get("url")
    if query.from_user.id not in user_passed:
        await query.answer("خاصك تدوز من الإعلان أولا!", show_alert=True)
        return
    await query.edit_message_text(f"⏳ كنحمل {'الفيديو HD' if mode=='video' else 'الصوت MP3'}... ممكن ياخد 20 ثانية")
    loop = asyncio.get_event_loop()
    path, title = await loop.run_in_executor(None, lambda: download_media(url, mode))
    if not path or not os.path.exists(path):
        await query.edit_message_text("❌ فشل التحميل، جرب رابط آخر (تيكتوك مضمون)")
        return
    try:
        if mode=="video":
            await ctx.bot.send_video(chat_id=query.message.chat_id, video=open(path,'rb'), caption=f"✅ {title}\nvia @Rapide Download")
        else:
            await ctx.bot.send_audio(chat_id=query.message.chat_id, audio=open(path,'rb'), title=title)
    except Exception as e:
        await ctx.bot.send_document(chat_id=query.message.chat_id, document=open(path,'rb'))
    finally:
        try: os.remove(path)
        except: pass
    await query.edit_message_text("✅ تم التحميل! صيفط رابط آخر")

async def dl_hd(update, ctx): await do_download(update, ctx, "video")
async def dl_mp3(update, ctx): await do_download(update, ctx, "audio")

def main():
    app = Application.builder().token(BOT_TOKEN).build()
    app.add_handler(CommandHandler("start", start))
    app.add_handler(CallbackQueryHandler(check_ad, pattern="check_ad"))
    app.add_handler(CallbackQueryHandler(dl_hd, pattern="dl_hd"))
    app.add_handler(CallbackQueryHandler(dl_mp3, pattern="dl_mp3"))
    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_link))
    app.run_polling()

if __name__ == "__main__":
    main()
